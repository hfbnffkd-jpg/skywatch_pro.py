# ============================================================
MIDEAST_BOUNDS = {
    "lat_min": 12.0,
    "lat_max": 42.0,
    "lon_min": 25.0,
    "lon_max": 60.0
}

# ============================================================
# وحدات المساعدة (Helpers)
# ============================================================
@dataclass
class Flight:
    icao24: str
    callsign: str
    origin_country: str
    time_position: float
    last_update: float
    lon: float
    lat: float
    altitude: Optional[float]
    velocity: Optional[float]
    heading: Optional[float]
    vertical_rate: Optional[float]
    squawk: Optional[str]
    spi: Optional[bool]
    position_source: Optional[int]
    category: Optional[int]

    def __post_init__(self):
        self.altitude = self.altitude if self.altitude is not None else 0.0
        self.velocity = self.velocity if self.velocity is not None else 0.0
        self.heading = self.heading if self.heading is not None else 0.0
        self.vertical_rate = self.vertical_rate if self.vertical_rate is not None else 0.0

    @property
    def is_high_altitude(self) -> bool:
        return self.altitude > 35000

    @property
    def is_fast(self) -> bool:
        return self.velocity > 500

    @property
    def is_unknown(self) -> bool:
        return self.origin_country == "UNKNOWN" or self.callsign == "UNKNOWN"

    @property
    def is_in_mideast(self) -> bool:
        return (MIDEAST_BOUNDS["lat_min"] <= self.lat <= MIDEAST_BOUNDS["lat_max"] and
                MIDEAST_BOUNDS["lon_min"] <= self.lon <= MIDEAST_BOUNDS["lon_max"])

    def get_aircraft_info(self) -> Dict:
        """استرجاع معلومات الطائرة من قاعدة البيانات"""
        # محاولة مطابقة النوع من الـ ICAO24 أو النداء
        # (في الواقع يتم المطابقة باستخدام قاعدة بيانات الـ ICAO)
        for key, info in AIRCRAFT_DB.items():
            if key in self.callsign or key in self.icao24:
                return info
        return {"type": "غير معروف", "country": "غير معروف", "icon": "✈️"}

    def get_vip_info(self) -> Optional[Dict]:
        """التحقق من كبار الشخصيات"""
        if self.icao24 in VIP_HEX:
            return VIP_HEX[self.icao24]
        return None

# المحرك الرئيسي: SKYWATCH PRO
# ============================================================
class SkyWatchPro:
    def __init__(self, config: Dict = None):
        self.config = config or {}
        self.session = requests.Session()
        self.session.headers.update({"User-Agent": "SkyWatchPro/3.0"})
        self.alerts = []
        self.flight_cache = {}
        self.db_conn = sqlite3.connect("skywatch.db")
        self.db_cursor = self.db_conn.cursor()
        self.init_db()
        self.running = True
        
        # إعدادات البوت
        self.telegram_token = self.config.get("telegram_token")
        self.telegram_chat_id = self.config.get("telegram_chat_id")
        
        # إعدادات المصادر
        self.sources = [
            {"name": "OpenSky", "url": "https://opensky-network.org/api/states/all"},
            # يمكن إضافة ADSBexchange API هنا عند الحاجة
        ]

    def init_db(self):
        """تهيئة قاعدة البيانات لتخزين السجلات"""
        self.db_cursor.execute("""
            CREATE TABLE IF NOT EXISTS flights (
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                icao24 TEXT,
                callsign TEXT,
                country TEXT,
                lat REAL,
                lon REAL,
                altitude REAL,
                velocity REAL,
                heading REAL,
                detected_at TEXT,
                alert_type TEXT,
                processed INTEGER DEFAULT 0
            )
        """)
        self.db_conn.commit()

    def fetch_flights(self) -> List[Flight]:
        """جلب البيانات من OpenSky"""
        try:
            resp = self.session.get("https://opensky-network.org/api/states/all", timeout=10)
            if resp.status_code == 200:
                data = resp.json()
                states = data.get('states', [])
                flights = []
                for state in states:
                    try:
                        flight = Flight(
                            icao24=state[0],
                            callsign=state[1] or "UNKNOWN",
                            origin_country=state[2] or "UNKNOWN",
                            time_position=state[3] or 0,
                            last_update=state[4] or 0,
                            lon=state[5] or 0.0,
                            lat=state[6] or 0.0,
                            altitude=state[7] or 0.0,
                            velocity=state[9] or 0.0,
                            heading=state[10] or 0.0,
                            vertical_rate=state[11] or 0.0,
                            squawk=state[13] or "0000",
                            spi=state[14] or False,
                            position_source=state[15] or 0,
                            category=state[16] or 0
                        )
                        if flight.is_in_mideast:
                            flights.append(flight)
                    except:
                        continue
                return flights
            else:
                return []
        except Exception as e:
            print(f"[!] خطأ في جلب البيانات: {e}")
            return []

    def analyze_flight(self, flight: Flight) -> List[Dict]:
        """تحليل الطائرة وكشف الأنشطة المشبوهة"""
        alerts = []
        aircraft_info = flight.get_aircraft_info()
        vip_info = flight.get_vip_info()

        # 1. الكشف عن كبار الشخصيات (VIP)
        if vip_info:
            alerts.append({
                "type": "VIP",
                "severity": "HIGH",
                "message": f"👑 طائرة كبار شخصيات: {vip_info['name']} ({vip_info['country']})",
                "data": vip_info
            })

        # 2. الكشف عن طائرات الاستطلاع
        if aircraft_info.get("type") in ["استطلاع إلكتروني", "استطلاع بحري", "إنذار مبكر"]:
            alerts.append({
                "type": "ISR",
                "severity": "HIGH",
                "message": f"🕵️ طائرة استطلاع/مراقبة: {aircraft_info['type']} ({aircraft_info['country']})",
                "data": aircraft_info
            })

        # 3. الكشف عن التزود بالوقود
        if "تزود بالوقود" in aircraft_info.get("type", ""):
            alerts.append({
                "type": "TANKER",
                "severity": "HIGH",
                "message": f"⛽ طائرة تزود بالوقود: {aircraft_info['type']}",
                "data": aircraft_info
            })

        # 4. الكشف عن النقل العسكري الثقيل
        if "نقل" in aircraft_info.get("type", "") and aircraft_info.get("country") in ["USA", "Russia", "UK"]:
            alerts.append({
                "type": "MILITARY_TRANSPORT",
                "severity": "MEDIUM",
                "message": f"📦 طائرة نقل عسكري: {aircraft_info['type']} ({aircraft_info['country']})",
                "data": aircraft_info
            })

        # 5. الكشف عن الطائرات بدون تعريف
        if flight.is_unknown:
            alerts.append({
                "type": "UNKNOWN",
                "severity": "HIGH",
                "message": f"🕵️ طائرة مجهولة الهوية: ICAO {flight.icao24}",
                "data": {"icao": flight.icao24, "callsign": flight.callsign}
            })

        # 6. الكشف عن الطائرات السريعة جداً (مقاتلات)
        if flight.is_fast:
            alerts.append({
                "type": "FAST_JET",
                "severity": "HIGH",
                "message": f"⚡ طائرة حربية سريعة: {flight.callsign} بسرعة {flight.velocity} عقدة",
                "data": {"speed": flight.velocity, "altitude": flight.altitude}
            })

        # 7. الكشف عن التحليق الدائري (استطلاع)
        # هنا يمكن إضافة خوارزمية تتبع المسار، لكن حالياً نستخدم ارتفاع ثابت وسرعة منخفضة
        if flight.altitude > 25000 and 200 < flight.velocity < 300:
            alerts.append({
                "type": "ORBIT",
                "severity": "MEDIUM",
                "message": f"🔄 نمط تحليق دائري (استطلاع محتمل): {flight.callsign}",
                "data": {"altitude": flight.altitude, "velocity": flight.velocity}
            })

        # 8. الكشف عن الطائرات في مناطق النزاع (حدود ساخنة)
        hot_zones = [
            {"name": "الخليج العربي", "lat_min": 26, "lat_max": 30, "lon_min": 48, "lon_max": 52},
            {"name": "سوريا/لبنان", "lat_min": 32, "lat_max": 34, "lon_min": 34, "lon_max": 36},
            {"name": "باب المندب", "lat_min": 12, "lat_max": 14, "lon_min": 43, "lon_max": 45},
        ]
        for zone in hot_zones:
            if (zone["lat_min"] <= flight.lat <= zone["lat_max"] and
                zone["lon_min"] <= flight.lon <= zone["lon_max"]):
                alerts.append({
                    "type": "HOT_ZONE",
                    "severity": "HIGH",
                    "message": f"🔥 طائرة في منطقة نزاع: {zone['name']} - {flight.callsign}",
                    "data": {"zone": zone["name"], "lat": flight.lat, "lon": flight.lon}
                })
                break

        # 9. الكشف عن الطائرات التي تغير ارتفاعها بشكل مفاجئ (تجنب رادار)
        if abs(flight.vertical_rate) > 3000:
            alerts.append({
                "type": "RAPID_CLIMB",
                "severity": "MEDIUM",
                "message": f"📈 تغير ارتفاع مفاجئ: {flight.callsign} بمعدل {flight.vertical_rate} قدم/دقيقة",
                "data": {"vertical_rate": flight.vertical_rate}
            })

        return alerts

    def enrich_alert(self, flight: Flight, alert: Dict) -> Dict:
        """إثراء التنبيه بمعلومات إضافية (صور، موقع)"""
        # جلب صورة الطائرة (محاكاة)
        image_url = "https://via.placeholder.com/400x200?text=No+Image"
        
        # تحديد الموقع الجغرافي
        location = f"{flight.lat:.4f}, {flight.lon:.4f}"
        
        # إضافة خريطة OpenStreetMap (رابط تتبع)
        map_link = f"https://www.openstreetmap.org/?mlat={flight.lat}&mlon={flight.lon}&zoom=8"
        
        enriched = {
            **alert,
            "image_url": image_url,
            "location": location,
            "map_link": map_link,
            "flight_data": {
                "callsign": flight.callsign,
                "icao": flight.icao24,
                "altitude": flight.altitude,
                "velocity": flight.velocity,
                "heading": flight.heading,
                "origin": flight.origin_country,
                "squawk": flight.squawk
            }
        }
        return enriched

    def send_telegram_alert(self, alert: Dict):
        """إرسال تنبيه إلى تليجرام"""
        if not self.telegram_token or not self.telegram_chat_id:
            print("[!] لم يتم إعداد التليجرام")
            return

        try:
            import asyncio
            from telegram import Bot
            
            async def send():
                bot = Bot(token=self.telegram_token)
                caption = (
                    f"🚨 **تنبيه استخباراتي جوي**\n\n"
                    f"**النوع:** {alert['type']}\n"
                    f"**الخطورة:** {alert['severity']}\n"
                    f"**الرسالة:** {alert['message']}\n"
                    f"**الموقع:** {alert.get('location', 'غير معروف')}\n"
                    f"**الارتفاع:** {alert.get('flight_data', {}).get('altitude', 0)} قدم\n"
                    f"**السرعة:** {alert.get('flight_data', {}).get('velocity', 0)} عقدة\n"
                    f"**النداء:** {alert.get('flight_data', {}).get('callsign', 'غير معروف')}\n"
                    f"**النداء:** {alert.get('flight_data', {}).get('icao', 'غير معروف')}\n\n"
                    f"[🗺️ عرض على الخريطة]({alert.get('map_link', '#')})"
                )
                await bot.send_message(chat_id=self.telegram_chat_id, text=caption, parse_mode="Markdown")
            
            asyncio.run(send())
        except Exception as e:
            print(f"[!] فشل إرسال التنبيه إلى تليجرام: {e}")

    def save_alert_to_db(self, flight: Flight, alert: Dict):
        """حفظ التنبيه في قاعدة البيانات"""
        try:
            self.db_cursor.execute("""
                INSERT INTO flights (icao24, callsign, country, lat, lon, altitude, velocity, heading, detected_at, alert_type)
                VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, ?)
            """, (
                flight.icao24,
                flight.callsign,
                flight.origin_country,
                flight.lat,
                flight.lon,
                flight.altitude,
                flight.velocity,
                flight.heading,
                datetime.now().isoformat(),
                alert['type']
            ))
            self.db_conn.commit()
        except Exception as e:
            print(f"[!] خطأ في حفظ البيانات: {e}")

    def run_cycle(self):
        """دورة تحليل واحدة"""
        print(f"[*] جلب البيانات من المصادر...")
        flights = self.fetch_flights()
        print(f"[+] تم جلب {len(flights)} طائرة في منطقة الشرق الأوسط")

        alert_count = 0
        for flight in flights:
            alerts = self.analyze_flight(flight)
            for alert in alerts:
                enriched = self.enrich_alert(flight, alert)
                alert_count += 1
                print(f"[🚨] {alert_count}: {alert['message']}")
                self.save_alert_to_db(flight, alert)
                self.send_telegram_alert(enriched)
        
        if alert_count == 0:
            print("[+] لا توجد أنشطة مشبوهة حالياً")
        else:
            print(f"[+] تم رصد {alert_count} نشاط مشبوه")

    def run_forever(self, interval: int = 30):
        """تشغيل المراقبة المستمرة"""
        print("\n🛰️ [غرفة العمليات] بدء المراقبة المستمرة للشرق الأوسط...")
        print(f"⏱️ التحديث كل {interval} ثانية")
        print("="*70)
        
        while self.running:
            try:
                self.run_cycle()
                time.sleep(interval)
            except KeyboardInterrupt:
                print("\n[+] إيقاف المراقبة...")
                self.running = False
                break
            except Exception as e:
                print(f"[!] خطأ في الدورة: {e}")
                time.sleep(10)
